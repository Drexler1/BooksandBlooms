# DAILY USE
docker compose up -d      # start
docker compose down       # stop

#AFTER CHANGING app.py
docker compose restart app

#AFTER CHANGING requirements.txt
docker compose up --build

#DATABASE 
# Connect to database
docker exec -it codecrew-db-1 mysql -uroot -pyourpassword pos_system

# Sync/export database
docker exec codecrew-db-1 mysqldump -uroot -pyourpassword pos_system > database/pos_system.sql

# Reset database with new sql file
docker compose down -v
docker compose up --build
